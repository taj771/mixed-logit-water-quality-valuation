function(apollo_beta, apollo_inputs, functionality = "estimate") {
  
  # Attach inputs
  apollo_attach(apollo_beta, apollo_inputs)
  on.exit(apollo_detach(apollo_beta, apollo_inputs))
  
  # Create list of probabilities
  P = list()
  
  # Define utilities
  V = list()
  V[["policy"]] = b_asc + 
    b_cost * COST + 
    
    # WQ improvements (unchanged)
    b_wq_local_basin * WQ_BASIN_LOCAL_POLICY +
    b_wq_nonlocal_basin * WQ_BASIN_NL_POLICY +
    b_wq_local_sub_basin * WQ_SUBBASIN_LOCAL_POLICY_SUBONLY +
    b_wq_nonlocal_sub_basin * WQ_SUBBASIN_NL_POLICY_SUBONLY +
    
    b_basewq_local*WQ_LOCAL_CHOICE_CURRENT*WQ_LOCAL_CHOICE_POLICY +
    b_basewq_nonlocal*WQ_NONLOCAL_CHOICE_CURRENT*WQ_NONLOCAL_CHOICE_POLICY +
    
    b_wq_x_bl_local*SQ_VARIATION_LOCAL_CHOICE*WQ_LOCAL_CHOICE_POLICY +
    b_wq_x_bl_nonlocal*SQ_VARIATION_NONLOCAL_CHOICE*WQ_NONLOCAL_CHOICE_POLICY 
  
  
  V[["opt_out"]] = 
    b_wq_local_basin * WQ_BASIN_LOCAL_CURRENT +
    b_wq_nonlocal_basin * WQ_BASIN_NL_CURRENT +
    b_wq_local_sub_basin * WQ_SUBBASIN_LOCAL_CURRENT_SUBONLY +
    b_wq_nonlocal_sub_basin * WQ_SUBBASIN_NL_CURRENT_SUBONLY +
    
    b_basewq_local*WQ_LOCAL_CHOICE_CURRENT*WQ_LOCAL_CHOICE_CURRENT +
    b_basewq_nonlocal*WQ_NONLOCAL_CHOICE_CURRENT*WQ_NONLOCAL_CHOICE_CURRENT +
    
    b_wq_x_bl_local*SQ_VARIATION_LOCAL_CHOICE*WQ_LOCAL_CHOICE_CURRENT +
    b_wq_x_bl_nonlocal*SQ_VARIATION_NONLOCAL_CHOICE*WQ_NONLOCAL_CHOICE_CURRENT 
  
  
  # Define MNL settings
  mnl_settings = list(
    alternatives  = c(policy = 1, opt_out = 0),  # Match to VOTE column coding
    avail         = 1,  # Both alternatives are always available
    choiceVar     = VOTE,  # Choice variable in the dataset
    utilities     = V
  )
  
  ### Compute probabilities using MNL model
  P[["model"]] = apollo_mnl(mnl_settings, functionality)

  P = apollo_prepareProb(P, apollo_inputs, functionality)
