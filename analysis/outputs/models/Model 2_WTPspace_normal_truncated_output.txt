function(apollo_beta, apollo_inputs, functionality = "estimate") {

  apollo_attach(apollo_beta, apollo_inputs)
  on.exit(apollo_detach(apollo_beta, apollo_inputs))

  P = list()
  
  # Define utilities - WTP SPACE
  V = list()
  V[["policy"]] = b_asc + 
    b_cost * (COST + 
                wtp_wq_local_basin * WQ_BASIN_LOCAL_POLICY +
                wtp_wq_nonlocal_basin * WQ_BASIN_NL_POLICY +
                wtp_wq_local_sub_basin * WQ_SUBBASIN_LOCAL_POLICY_SUBONLY +
                wtp_wq_nonlocal_sub_basin * WQ_SUBBASIN_NL_POLICY_SUBONLY)
  
  # Opt-out alternative
  V[["opt_out"]] = 
    b_cost * (wtp_wq_local_basin * WQ_BASIN_LOCAL_CURRENT +
                wtp_wq_nonlocal_basin * WQ_BASIN_NL_CURRENT +
                wtp_wq_local_sub_basin * WQ_SUBBASIN_LOCAL_CURRENT_SUBONLY +
                wtp_wq_nonlocal_sub_basin * WQ_SUBBASIN_NL_CURRENT_SUBONLY)

    alternatives  = c(policy = 1, opt_out = 0),

  P[["model"]] = apollo_mnl(mnl_settings, functionality)
